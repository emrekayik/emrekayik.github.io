<script src="/assets/lib/gsap.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/ScrollTrigger.min.js"></script>

<script>
    document.addEventListener("DOMContentLoaded", function() {
        if (typeof gsap !== "undefined") {
            gsap.registerPlugin(ScrollTrigger);

            gsap.from("article > *", {
                opacity: 0,
                y: 20,
                duration: 0.8,
                stagger: 0.1,
                ease: "power2.out"
            });
        }
    });

    function animateBasics() {
        gsap.to("#box1", { x: 50, duration: 1, repeat: 1, yoyo: true, ease: "back.inOut" });
        gsap.to("#box2", { rotation: 360, duration: 1, repeat: 1, yoyo: true });
        gsap.to("#box3", { scale: 1.5, duration: 1, repeat: 1, yoyo: true, ease: "elastic.out(1, 0.3)" });
    }

    function playTimeline() {
        const tl = gsap.timeline();
        tl.to(".tl-box", { x: "100%", duration: 0.5, stagger: 0.2, ease: "power1.inOut" })
          .to(".tl-box", { rotation: 360, duration: 0.5, stagger: 0.1 })
          .to(".tl-box", { x: 0, duration: 0.5, stagger: -0.2, ease: "back.out" });
    }

    function animateStagger() {
        gsap.fromTo(".stagger-item", 
            { scale: 1, opacity: 1 },
            { 
                scale: 0.1, 
                opacity: 0.3, 
                duration: 0.5, 
                stagger: {
                    each: 0.1,
                    from: "center",
                    grid: [3, 5]
                },
                yoyo: true,
                repeat: 1
            }
        );
    }
</script>
